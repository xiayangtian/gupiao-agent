```python
    def _extract_metrics(
        self, pdf_text: str, model: Optional[str] = None
    ) -> Optional[List[Dict[str, Any]]]:
        """
        从财报文本中抽取结构化指标（折线图数据源）。

        一次结构化调用（response_format=json_object，max_tokens=2000）。
        任何失败（网络/非法 JSON/解析失败）只返回 None，不抛异常。

        Returns:
            [{"year": int, "revenue": float|None, "net_profit": float|None,
              "roe": float|None, "gross_margin": float|None,
              "debt_ratio": float|None}, ...] 按年升序；数据全部为空返回 None。
        """

        def _num(value: Any) -> Optional[float]:
            """尽力转数字：None/空串/非法字符串/非有限值 → None；字符串数字 → float"""
            if value is None or value == "":
                return None
            try:
                value = float(value)
            except (TypeError, ValueError):
                return None
            if value != value or value in (float("inf"), float("-inf")):
                return None
            return value

        try:
            resp = self.client.chat(
                messages=[{
                    "role": "user",
                    "content": f"{METRICS_PROMPT}\n\n--- 财报内容开始 ---\n"
                    f"{pdf_text}\n--- 财报内容结束 ---",
                }],
                model=model,
                max_tokens=2000,
                response_format={"type": "json_object"},
            )
            payload = json.loads(resp["content"])
        except Exception as exc:
            logger.warning("指标抽取失败：%s", exc)
            return None

        if not isinstance(payload, dict):
            return None
        rows = payload.get("metrics")
        if not isinstance(rows, list):
            return None

        now_year = datetime.date.today().year
        by_year: Dict[int, Dict[str, Any]] = {}
        for row in rows:
            if not isinstance(row, dict):
                continue
            try:
                year = int(row.get("year"))
            except (TypeError, ValueError):
                continue
            if not (2000 <= year <= now_year + 1):
                continue
            # 同一个年份重复出现 → 后写覆盖前写（保持升序 dict 键）
            by_year[year] = {
                "year": year,
                "revenue": _num(row.get("revenue")),
                "net_profit": _num(row.get("net_profit")),
                "roe": _num(row.get("roe")),
                "gross_margin": _num(row.get("gross_margin")),
                "debt_ratio": _num(row.get("debt_ratio")),
            }

        result = [by_year[y] for y in sorted(by_year)]
        return result or None
```

```python
"""webapp.history — 本地分析产物与已下载 PDF 的扫描/聚合

供「分析历史」入口（GET /api/history）与财报列表「已分析」标记使用。
纯函数设计（不依赖 FastAPI/磁盘以外的状态），便于单元测试。

文件命名约定（与 downloader.build_filename / AnalysisReport.save 对齐）：
    PDF:   {公司}_{代码}_{年报|半年报|季报}_{年份}.pdf
    分析:  {公司}_{代码}_{年份}_分析报告.json   （Web，safe_company 含 6 位代码）
           {公司}_{年份}_分析报告.json          （旧 CLI，无代码）
"""

import json
import logging
import os
import re
from typing import Any, Dict, List, Optional

logger = logging.getLogger(__name__)

PDF_NAME_RE = re.compile(
    r"^(?P<rest>.+?)_(?P<code>\d{6})_(?P<kind>年报|半年报|季报)_(?P<year>\d{4})\.pdf$"
)


def parse_analysis_filename(basename: str) -> Optional[Dict[str, Any]]:
    """
    从分析文件名解析 (company_label, code?, year)。

    两种形态：
        {company}_{code}_{year}_分析报告.json  → code/year
        {company}_{year}_分析报告.json         → year（无 code）
    无法解析年份 → None。
    """
    m = re.match(r"^(?P<name>.+?)_(?P<year>\d{4})_分析报告\.json$", basename)
    if not m:
        return None
    year = int(m.group("year"))
    name = m.group("name")
    code = None
    cm = re.search(r"(?P<code>\d{6})$", name)
    if cm:
        code = cm.group("code")
        name = name[: cm.start()].rstrip("_")
    return {"company_label": name, "code": code, "year": year}


def _read_analysis(path: str) -> Optional[Dict[str, Any]]:
    """读取分析 JSON；损坏返回 None（不影响其他条目）"""
    try:
        with open(path, "r", encoding="utf-8") as f:
            data = json.load(f)
        return data if isinstance(data, dict) else None
    except (json.JSONDecodeError, OSError, IOError):
        logger.warning("跳过无法解析的分析文件：%s", path)
        return None


def _analyze_entry(path: str) -> Optional[Dict[str, Any]]:
    """单份分析文件 → 历史条目；内容损坏返回 None"""
    basename = os.path.basename(path)
    content = _read_analysis(path)
    if content is None:
        return None
    parsed = parse_analysis_filename(basename)
    year = (parsed or {}).get("year")
    meta = content.get("meta") or {}
    period = meta.get("period") or None          # 新 Web 格式精确报告期
    if not period and year:                      # 旧文件回退：按年份推断年末
        period = f"{year}-12-31"
    try:
        mtime = os.path.getmtime(path)
    except OSError:
        mtime = 0.0
    return {
        "filename": basename,
        "company_label": (parsed or {}).get("company_label") or "",
        "code": (parsed or {}).get("code"),
        "year": year,
        "period": period,
        "period_exact": bool(meta.get("period")),
        "content": content,
        "mtime": mtime,
    }


def scan_locals(analysis_dir: str, pdf_dir: str) -> Dict[str, Any]:
    """
    扫描本地分析 JSON 与已下载 PDF，按公司聚合（无需 AI 调用）。

    Returns:
        {
            "companies": [{
                "code": str, "name": str, "mtime": float,
                "analyses": [《entry],  # 每个 entry: filename/code/company_label/year/period/period_exact/content/mtime
                "pdfs": [{"title", "period", "type", "clickable"}],
            }],
            "orphan_analyses": [无 code 的《_analyze_entry 条目],
        }
    排序：companies 按 mtime（组内最新分析）倒序；analyses 年份倒序；pdfs period 倒序。
    """
    companies: Dict[str, Dict[str, Any]] = {}
    orphans: List[Dict[str, Any]] = []

    # 1) 分析文件
    if os.path.isdir(analysis_dir):
        for fname in sorted(os.listdir(analysis_dir)):
            if not fname.endswith("_分析报告.json"):
                continue
            entry = _analyze_entry(os.path.join(analysis_dir, fname))
            if entry is None:
                continue
            if entry["code"]:
                c = companies.setdefault(
                    entry["code"],
                    {"code": entry["code"], "name": "", "mtime": 0.0,
                     "analyses": [], "pdfs": []},
                )
                c["analyses"].append(entry)
                c["mtime"] = max(c["mtime"], entry["mtime"])
            else:
                orphans.append(entry)

    # 2) 已下载 PDF
    if os.path.isdir(pdf_dir):
        for fname in sorted(os.listdir(pdf_dir)):
            m = PDF_NAME_RE.match(fname)
            if not m:
                continue
            kind = m.group("kind")
            year = int(m.group("year"))
            if kind == "年报":
                period, clickable = f"{year}-12-31", True
            else:  # 半年报/季报无法确定精确期 → 仅展示不可点
                period, clickable = None, False
            companies.setdefault(
                m.group("code"),
                {"code": m.group("code"), "name": "", "mtime": 0.0,
                 "analyses": [], "pdfs": []},
            )["pdfs"].append(
                {"title": fname, "period": period, "type": kind, "clickable": clickable}
            )

    # 3) 公司名（取组内最新分析条目的 meta.company，否则留空）+ 排序
    result = []
    for c in companies.values():
        if c["analyses"]:
            newest = sorted(c["analyses"], key=lambda a: a["mtime"], reverse=True)[0]
            c["name"] = newest["content"].get("meta", {}).get("company") or newest["name"] or ""
        c["analyses"].sort(key=lambda a: (a["year"] or 0), reverse=True)
        c["pdfs"].sort(key=lambda p: p["period"] or "0000-00-00", reverse=True)
        result.append(c)
    result.sort(key=lambda c: c["mtime"], reverse=True)

    return {"companies": result, "orphan_analyses": orphans}


def analyzed_map(analysis_dir: str) -> Dict[str, Dict[str, Any]]:
    """
    财报列表「已分析」标记用。

    Returns: {code: {"years": set(int), "periods": set(str)}}
    - periods：meta.period 精确匹配
    - years：文件名年份（近似匹配；旧格式文件无代码时无法归属，跳过）
    """
    out: Dict[str, Dict[str, Any]] = {}
    if not os.path.isdir(analysis_dir):
        return out
    for fname in sorted(os.listdir(analysis_dir)):
        if not fname.endswith("_分析报告.json"):
            continue
        content = _read_analysis(os.path.join(analysis_dir, fname))
        if content is None:
            continue
        parsed = parse_analysis_filename(fname)
        if not parsed or not parsed["code"]:
            continue
        slot = out.setdefault(parsed["code"], {"years": set(), "periods": set()})
        if parsed["year"]:
            slot["years"].add(parsed["year"])
        period = (content.get("meta") or {}).get("period")
        if period:
            slot["periods"].add(period)
    return out
```

function mergeYearMetrics(rows) {
